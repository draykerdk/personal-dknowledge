## 📡 Radar das 12 semanas

> *Uma linha por semana do trimestre. A semana ainda sem nota aparece como "não criada", e a falta de registro não conta contra nada.*

```dataviewjs
try {
  const cur = dv.current();
  const quarter = String(cur.quarter || (String(cur.file.name).match(/\d{4}-Q[1-4]/) || [])[0] || '');
  const m = quarter.match(/(\d{4})-Q([1-4])/);
  if (!m) {
    dv.paragraph('*Preencha `quarter` nas propriedades desta nota para ver o radar.*');
  } else {
    const ano = Number(m[1]), q = Number(m[2]);
    const mesInicial = (q - 1) * 3 + 1;
    const inicio = moment(`${ano}-${String(mesInicial).padStart(2, '0')}-01`, 'YYYY-MM-DD').startOf('isoWeek');
    const tasks = dv.pages('"TaskNotes"');
    // As SMART e as ações de cada semana pela forma de mostrar do plugin, no horizonte dela.
    const nucleo = app.plugins.plugins['dfm-system']?.nucleo;
    const ehSmart = t => String(t.type ?? '') === 'tarefa_smart';
    const smartTodas = tasks.where(t => ehSmart(t));
    const acoesTodas = tasks.where(t => !ehSmart(t));
    const forma = (itens, semana) => {
      const f = nucleo?.formaDasTarefas?.(itens, semana, '—');
      return f ? f.text : 'sem leitura';
    };
    // O ajuste mínimo de cada semana vem da revisão dela (400 - MEMORY/200 - WEEKLY, campo `week`).
    const revisoes = dv.pages('"400 - MEMORY/200 - WEEKLY"').where(p => String(p.type ?? '') === 'review_weekly');
    const linhas = [];
    for (let i = 0; i < 14; i++) {
      const data = moment(inicio).add(i, 'weeks');
      // A semana pertence ao trimestre em que cai a quinta-feira dela.
      const quinta = moment(data).add(3, 'days');
      const mes = quinta.month() + 1;
      if (quinta.year() !== ano || mes < mesInicial || mes >= mesInicial + 3) continue;
      const nome = data.format('GGGG-[W]WW');
      const caminho = `300 - RUNTIME/302 - Daymiss/Week/${nome}.md`;
      const semana = dv.page(caminho);
      const revisao = revisoes.where(r => String(r.week ?? '') === nome).first();
      const ajuste = (revisao && revisao.ajuste_minimo) || (semana && semana.ajuste_minimo) || '—';
      linhas.push([
        semana ? `[[${caminho}|${nome}]]` : nome,
        semana ? (semana.foco || '—') : '*não criada*',
        forma(smartTodas, nome),
        forma(acoesTodas, nome),
        ajuste,
        semana ? (semana.status || 'open') : '—'
      ]);
    }
    dv.table(['Semana', 'Fio condutor', 'SMART', 'Ações', 'Ajuste mínimo', 'Status'], linhas);
  }
} catch (e) { dv.paragraph('—'); }
```

## 🌟 Temporada

> *As centrais do trimestre (`temporada`) e, abaixo de cada uma, os auxiliares confirmados no 🧭 (`projetos_foco`). O auxiliar aponta para a central no campo `projects` dele. Até três centrais.*

```dataviewjs
try {
  const cur = dv.current();
  const quarter = String(cur.quarter || (String(cur.file.name).match(/\d{4}-Q[1-4]/) || [])[0] || '');
  const nucleo = app.plugins.plugins['dfm-system']?.nucleo;
  const dobrar = s => String(s ?? '').normalize('NFD').replace(/[̀-ͯ]/g, '').toLowerCase().trim();
  // Os nomes de um campo de links: a lista, o link do Dataview, o texto "[[A]], [[B]]" ou o nome solto.
  const nomes = (valor) => {
    const itens = Array.isArray(valor) ? valor
      : (valor && typeof valor === 'object' && !valor.path && typeof valor[Symbol.iterator] === 'function') ? [...valor]
      : (valor ? [valor] : []);
    const out = [];
    for (const item of itens) {
      const texto = item?.path ?? String(item ?? '');
      const links = [...texto.matchAll(/\[\[([^\]|#]+)/g)].map(m => m[1]);
      for (const bruto of (links.length ? links : [texto])) {
        const nome = String(bruto).split('/').pop().replace(/\.md$/, '').trim();
        if (nome && !out.some(n => dobrar(n) === dobrar(nome))) out.push(nome);
      }
    }
    return out;
  };
  const centrais = nomes(cur.temporada);
  if (!centrais.length) {
    dv.paragraph('*Ainda sem Temporada: o 🧭 Planejar o trimestre pergunta as centrais.*');
  } else {
    // A nota-índice que se chama README é nomeada pela pasta dela.
    const rotulo = (p) => p.file.name === 'README' ? p.file.folder.split('/').pop() : p.file.name;
    // A coluna do quadro do projeto (o número da pasta); departamento fica no fim e o encerrado, de fora.
    const coluna = (p) => (p.file.path.match(/\/(\d) - [^/]+\//) || [])[1] ?? '9';
    const ordem = { '2': 0, '3': 1, '4': 2, '1': 5 };
    const notas = dv.pages('"100 - AREAS"')
      .where(p => ['iniciativa', 'projeto', 'departamento'].includes(String(p.type ?? '')) && !['5', '6'].includes(coluna(p)));
    const formaDe = (nome) => nucleo?.formaDoProjeto?.(nome, quarter)?.text ?? 'sem leitura (o dfm-system não está carregado)';
    const mesmo = (p, nome) => dobrar(rotulo(p)) === dobrar(nome) || dobrar(p.file.name) === dobrar(nome);
    const ehCentral = (p) => centrais.some(c => mesmo(p, c));
    const serveA = (p, central) => nomes(p.projects).some(n => dobrar(n) === dobrar(central));
    // A lista confirmada no 🧭; vazia, entram todos os que apontam para uma central.
    const foco = nomes(cur.projetos_foco);
    const confirmado = (p) => !foco.length || foco.some(n => mesmo(p, n));
    const linhas = [];
    const fora = [];
    for (const central of centrais) {
      const nota = notas.find(p => mesmo(p, central));
      const nome = nota ? rotulo(nota) : central;
      linhas.push([`🌟 ${nota ? `[[${nota.file.path}|${nome}]]` : nome}`, 'central', formaDe(nome)]);
      const ligados = notas
        .where(p => !ehCentral(p) && serveA(p, central))
        .array()
        .sort((a, b) => ((ordem[coluna(a)] ?? 9) - (ordem[coluna(b)] ?? 9)) || rotulo(a).localeCompare(rotulo(b), 'pt-BR'));
      for (const p of ligados) {
        if (!confirmado(p)) { if (!fora.includes(rotulo(p))) fora.push(rotulo(p)); continue; }
        linhas.push([`↳ [[${p.file.path}|${rotulo(p)}]]`, `auxiliar: serve a ${nome}`, formaDe(rotulo(p))]);
      }
    }
    // Na lista do 🧭 sem apontar para nenhuma central: aparece, com o que falta.
    for (const n of foco) {
      if (centrais.some(c => dobrar(c) === dobrar(n))) continue;
      const nota = notas.find(p => mesmo(p, n));
      if (nota && centrais.some(c => serveA(nota, c))) continue;
      const nome = nota ? rotulo(nota) : n;
      linhas.push([`↳ ${nota ? `[[${nota.file.path}|${nome}]]` : nome}`, 'auxiliar sem central: ligue pelo `projects` ou tire no 🧭', formaDe(nome)]);
    }
    dv.table(['Projeto', 'Papel', 'No trimestre'], linhas);
    if (fora.length) dv.paragraph(`*Apontam para uma central, mas não estão na lista do trimestre (\`projetos_foco\`): ${fora.join(' · ')}.*`);
  }
} catch (e) { dv.paragraph('—'); }
```

```dataviewjs
try {
  const cur = dv.current();
  const quarter = String(cur.quarter || (String(cur.file.name).match(/\d{4}-Q[1-4]/) || [])[0] || '');
  const dobrar = s => String(s ?? '').normalize('NFD').replace(/[̀-ͯ]/g, '').toLowerCase().trim();
```

```dataviewjs
try {
  const cur = dv.current();
  const quarter = String(cur.quarter || (String(cur.file.name).match(/\d{4}-Q[1-4]/) || [])[0] || '');
  const m = quarter.match(/(\d{4})-Q([1-4])/);
  if (!m) {
    dv.paragraph('*Preencha `quarter` nas propriedades para calcular as semanas das paradas.*');
  } else {
    const ano = Number(m[1]), q = Number(m[2]);
    const mesInicial = (q - 1) * 3 + 1;
    const inicio = moment(`${ano}-${String(mesInicial).padStart(2, '0')}-01`, 'YYYY-MM-DD').startOf('isoWeek');
    const semanas = [];
    for (let i = 0; i < 14; i++) {
      const data = moment(inicio).add(i, 'weeks');
      const quinta = moment(data).add(3, 'days');
      const mes = quinta.month() + 1;
      if (quinta.year() !== ano || mes < mesInicial || mes >= mesInicial + 3) continue;
      semanas.push(data);
    }
    const paradas = [
      [4, 'O marco das semanas 1–4 aconteceu?'],
      [8, 'O marco das semanas 5–8 aconteceu? O que sai para o último caber?'],
      [12, 'O que fecha neste trimestre e o que passa para o próximo?']
    ];

      const data = semanas[n - 1];
      if (!data) return [`${n}ª`, '—', pergunta];
      const nome = data.format('GGGG-[W]WW');
      const caminho = `300 - RUNTIME/302 - Daymiss/Week/${nome}.md`;
      const rotulo = `${nome} (${data.format('DD/MM')}–${moment(data).add(6, 'days').format('DD/MM')})`;
      return [`${n}ª`, dv.page(caminho) ? `[[${caminho}|${rotulo}]]` : rotulo, pergunta];
    }));
  }
} catch (e) { dv.paragraph('—'); }
```
